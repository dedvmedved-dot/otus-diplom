# TASK-P7B2-R1 — Hardened P7B2 final read-only verifier (portable).
# Uses shared logic ansible/tools/p7b2_verify_logic.py (copied to nodes).
# All freshness/recovery/top-pods gates are real. any_errors_fatal fail-closed.

- name: "P7B2 final read-only verifier (hardened)"
  hosts: all
  gather_facts: false
  any_errors_fatal: true

  tasks:
    - name: Copy shared validation logic to node-01
      ansible.builtin.copy:
        src: "{{ playbook_dir }}/../tools/p7b2_verify_logic.py"
        dest: /tmp/p7b2_verify_logic.py
        mode: 0644
      delegate_to: node-01
      run_once: true

    - name: P7B2 cluster + metrics + freshness + top validation (read-only, node-01)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import json, subprocess, sys
          sys.path.insert(0, "/tmp")
          import p7b2_verify_logic as V
          KC = ["kubectl", "--kubeconfig", "/etc/kubernetes/admin.conf"]
          def runj(c):
              r = subprocess.run(c, capture_output=True, text=True)
              if r.returncode != 0:
                  raise AssertionError(f"query failed: {' '.join(c)} rc={r.returncode}")
              return json.loads(r.stdout)

          nodes = runj(KC + ["get", "nodes", "-o", "json"])
          items = nodes.get("items"); assert isinstance(items, list)
          for it in items:
              assert it["status"]["nodeInfo"]["kubeletVersion"] == "v1.36.2", "k8s drift"
          print("P7B2_K8S_OK")
          assert len(items) == 3, f"node count {len(items)}"
          for it in items:
              assert any(c["type"]=="Ready" and c["status"]=="True" for c in it["status"]["conditions"])
          print("P7B2_NODES_READY")

          mc = runj(KC + ["-n","metallb-system","get","deploy","controller","-o","json"])
          assert mc["status"].get("availableReplicas",0)==1
          ms = runj(KC + ["-n","metallb-system","get","ds","speaker","-o","json"])
          assert ms["status"].get("numberReady",0)==3
          print("P7B2_P6B_PRESERVED")

          ed = runj(KC + ["-n","envoy-gateway-system","get","deploy","envoy-gateway","-o","json"])
          assert ed["status"].get("availableReplicas",0)==1
          gc = runj(KC + ["get","gatewayclass","eg","-o","json"])
          acc=[c["status"] for c in gc["status"]["conditions"] if c["type"]=="Accepted"]
          assert acc and acc[0]=="True"
          print("P7B2_P6C_PRESERVED")

          kvp = runj(KC + ["-n","kube-system","get","pod","kube-vip-node-01","-o","json"])
          env = {e["name"]: e.get("value","") for e in kvp["spec"]["containers"][0]["env"]}
          assert env.get("svc_enable")=="false" and env.get("address")=="172.30.140.100"
          gw = runj(KC + ["get","gateway","-A","-o","json"]); assert len(gw.get("items",[]))==0
          hr = runj(KC + ["get","httproute","-A","-o","json"]); assert len(hr.get("items",[]))==0
          print("P7B2_P6D_KUBECTL_OK")

          deps = runj(KC + ["get","deploy","-A","-o","json"])
          V.validate_velero([d["metadata"]["name"] for d in deps.get("items",[])])
          print("P7B2_VELERO_ABSENT")

          mdep = runj(KC + ["-n","kube-system","get","deploy","metrics-server","-o","json"])
          V.validate_deploy(mdep["status"].get("availableReplicas",0), mdep["spec"]["replicas"])
          print("P7B2_METRICS_SERVER_OK")

          pods = runj(KC + ["-n","kube-system","get","pods","-l","k8s-app=metrics-server","-o","json"])
          c0 = pods["items"][0]["spec"]["containers"][0]
          req_img = c0.get("image","")
          iid = pods["items"][0]["status"]["containerStatuses"][0]["imageID"]
          V.validate_image(req_img, iid)
          print("P7B2_IMAGE_DIGEST_OK")
          V.validate_args(c0.get("args", []))
          print("P7B2_INSECURE_FLAGS_ABSENT")

          api = runj(KC + ["get","apiservice","v1beta1.metrics.k8s.io","-o","json"])
          avail = [c["status"] for c in api["status"]["conditions"] if c["type"]=="Available"]
          V.validate_apiservice(avail[0] if avail else "False")
          print("P7B2_APISERVICE_OK")

          nm = runj(KC + ["get","--raw","/apis/metrics.k8s.io/v1beta1/nodes"])
          V.validate_nodemetrics(nm)
          print("P7B2_NODE_METRICS_OK")
          V.validate_freshness(nm)
          print("P7B2_METRICS_FRESHNESS_OK")

          top_nodes = subprocess.run(KC + ["top","nodes"], capture_output=True, text=True)
          assert top_nodes.returncode==0
          V.validate_top_nodes(top_nodes.stdout)
          print("P7B2_TOP_NODES_OK")

          top_pods = subprocess.run(KC + ["top","pods","-A"], capture_output=True, text=True)
          assert top_pods.returncode==0
          V.validate_top_pods(top_pods.stdout)
          print("P7B2_TOP_PODS_OK")

          csrs = runj(KC + ["get","csr","-o","json"])
          V.validate_csr(csrs.get("items", []))
          print("P7B2_CSR_STATE_OK")
          PYEOF
      changed_when: false
      delegate_to: node-01
      run_once: true

    - name: Copy shared validation logic to all nodes
      ansible.builtin.copy:
        src: "{{ playbook_dir }}/../tools/p7b2_verify_logic.py"
        dest: /tmp/p7b2_verify_logic.py
        mode: 0644

    - name: P7B2 node kubelet TLS + P6D route + P7A VLAN143 (read-only, all nodes)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import subprocess, socket, sys
          sys.path.insert(0, "/tmp")
          import p7b2_verify_logic as V
          host = socket.gethostname()
          cfg = open("/var/lib/kubelet/config.yaml").read()
          V.validate_kubelet_cfg(cfg)
          print("P7B2_SERVER_TLS_BOOTSTRAP_OK")
          print("P7B2_ROTATE_CERTIFICATES_OK")
          cert = "/var/lib/kubelet/pki/kubelet-server-current.pem"
          v = subprocess.run(["openssl","verify","-CAfile","/etc/kubernetes/pki/ca.crt",cert],capture_output=True,text=True)
          V.validate_ca_trust(v.returncode)
          print("P7B2_KUBELET_CLUSTER_CA_TRUST_OK")
          san = subprocess.run(["openssl","x509","-in",cert,"-noout","-ext","subjectAltName"],capture_output=True,text=True).stdout
          assert host in san, f"DNS SAN missing {host}"
          print("P7B2_KUBELET_DNS_SAN_OK")
          p = subprocess.run(["bash","-c","tr '\\0' '\\n' < /proc/$(pgrep -xo kubelet)/cmdline | grep -oE 'node-ip=[0-9.]+' | cut -d= -f2"],capture_output=True,text=True)
          ip = p.stdout.strip()
          assert ip and ip in san, f"IP SAN missing {ip}"
          print("P7B2_KUBELET_IP_SAN_OK")
          t = subprocess.run(["openssl","s_client","-connect",f"{ip}:10250","-verify_return_error","-CAfile","/etc/kubernetes/pki/ca.crt","-verify_ip",ip],input="",capture_output=True,text=True)
          assert t.returncode==0, "InternalIP TLS failed"
          print("P7B2_KUBELET_TLS_INTERNALIP_OK")
          h = subprocess.run(["openssl","s_client","-connect",f"{ip}:10250","-servername",host,"-verify_hostname",host,"-verify_return_error","-CAfile","/etc/kubernetes/pki/ca.crt"],input="",capture_output=True,text=True)
          assert h.returncode==0, "hostname TLS failed"
          print("P7B2_KUBELET_TLS_HOSTNAME_OK")
          rr = subprocess.run(["ip","route","show","172.30.20.0/24"],capture_output=True,text=True)
          V.validate_route(rr.stdout)
          print("P7B2_P6D_PRESERVED")
          b = subprocess.run(["ip","-br","addr","show","bond0.143"],capture_output=True,text=True)
          V.validate_vlan143(b.stdout)
          print("P7B2_P7A_PRESERVED")
          PYEOF
      changed_when: false

    - name: P7B2 README + recovery evidence + transition + final PASS (read-only, local)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import glob, os, subprocess, sys
          root = subprocess.run(["git","rev-parse","--show-toplevel"], capture_output=True, text=True).stdout.strip()
          sys.path.insert(0, os.path.join(root, "ansible", "tools"))
          import p7b2_verify_logic as V

          data = open(os.path.join(root, "README.md")).read()
          V.validate_readme(data)
          print("P7B2_README_STATUS_OK")

          # recovery evidence from committed P7B2 package (read-only)
          p7b2_dir = None
          for d in glob.glob(os.path.join(root, "evidence", "P7", "*task-p7b2-metrics-server-secure-deployment")):
              p7b2_dir = d
          assert p7b2_dir, "P7B2 evidence dir not found"
          def rd(fn):
              with open(os.path.join(p7b2_dir, fn)) as f:
                  return f.read()
          old_uid = ""
          new_uid = ""
          for line in rd("43-recovery-before.txt").splitlines():
              if "uid=" in line: old_uid = line.split("uid=",1)[1].split()[0]
          for line in rd("45-recovery-after.txt").splitlines():
              if "uid=" in line: new_uid = line.split("uid=",1)[1].split()[0]
          apisvc = rd("46-recovery-metrics-api.txt")
          top_out = rd("47-recovery-top-nodes.txt")
          V.validate_recovery(old_uid, new_uid,
                              "YES" if "APISERVICE_RECOVERED=YES" in apisvc else "NO",
                              "YES" if "METRICS_API_RECOVERED=YES" in apisvc else "NO",
                              top_out)
          print("P7B2_RECOVERY_UID_CHANGE_OK")
          print("P7B2_RECOVERY_APISERVICE_OK")
          print("P7B2_RECOVERY_METRICS_OK")
          print("P7B2_RECOVERY_TOP_NODES_OK")

          print("P7A_NO_METRICS_SERVER_ASSERTION=SUPERSEDED_BY_P7B2")
          print("P7B1_NO_METRICS_SERVER_ASSERTION=SUPERSEDED_BY_P7B2")
          print("P7B2_FINAL_READONLY_VERIFIER_PASS")
          PYEOF
      changed_when: false
      delegate_to: localhost
      connection: local
      run_once: true
