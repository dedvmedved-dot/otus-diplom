#!/usr/bin/env python3
"""
TASK-P7B1-R1 — Harden kubelet-serving CSR policy validation.

FAIL-CLOSED. Read-only. Kubernetes Node JSON is AUTHORITATIVE for the InternalIP.
Operator may only supply the CSR JSON, the live Node JSON, and optionally an
expected node name (assertion-only). Operator MUST NOT supply the InternalIP.

Usage:
  python3 p7b1_validate_kubelet_serving_csr.py <csr.json> --node-json <node.json> [--expected-node <name>]

Exit 0 only when ALL checks pass.
"""

import argparse
import base64
import ipaddress
import json
import sys

from cryptography import x509
from cryptography.x509.oid import NameOID

SIGNER = "kubernetes.io/kubelet-serving"
ALLOWED_USAGES = {"digital signature", "key encipherment", "server auth"}


def fail(msg):
    raise AssertionError(msg)


def load_node(node_path):
    with open(node_path) as f:
        node = json.load(f)
    if not isinstance(node, dict):
        fail("Node JSON must be an object")
    if node.get("kind") != "Node":
        fail(f"kind != Node: {node.get('kind')}")
    name = (node.get("metadata") or {}).get("name")
    if not name:
        fail("Node metadata.name missing")
    status = node.get("status") or {}
    conditions = status.get("conditions", [])
    ready = any(c.get("type") == "Ready" and c.get("status") == "True" for c in conditions)
    if not ready:
        fail(f"Node {name} not Ready")
    addresses = status.get("addresses", [])
    ips = [a["address"] for a in addresses if a.get("type") == "InternalIP"]
    if len(ips) == 0:
        fail(f"Node {name} has no InternalIP")
    if len(ips) > 1:
        fail(f"Node {name} has ambiguous InternalIPs: {ips}")
    ip = ips[0]
    try:
        ipaddress.ip_address(ip)
    except ValueError:
        fail(f"Node {name} InternalIP invalid: {ip}")
    return name, ip


def parse_csr(request_b64):
    """Parse PKCS#10 and return (cn, o, dns_sans, ip_sans, other_san_types)."""
    try:
        pem = base64.b64decode(request_b64)
    except Exception as e:
        fail(f"cannot base64-decode spec.request: {e}")
    try:
        csr = x509.load_pem_x509_csr(pem)
    except Exception as e:
        fail(f"cannot parse PKCS#10: {e}")

    cn = None
    o = None
    for attr in csr.subject:
        if attr.oid == NameOID.COMMON_NAME:
            cn = attr.value
        if attr.oid == NameOID.ORGANIZATION_NAME:
            o = attr.value

    dns_sans = []
    ip_sans = []
    other_sans = []
    try:
        san_ext = csr.extensions.get_extension_for_class(x509.SubjectAlternativeName).value
        for gn in san_ext._general_names:
            if isinstance(gn, x509.DNSName):
                dns_sans.append(gn.value)
            elif isinstance(gn, x509.IPAddress):
                ip_sans.append(str(gn.value))
            else:
                other_sans.append(type(gn).__name__)
    except x509.ExtensionNotFound:
        pass  # no SAN => empty lists (will fail missing DNS/IP later)

    return cn, o, dns_sans, ip_sans, other_sans


def validate(csr, live_node, live_ip, expected_node=None):
    if not isinstance(csr, dict):
        fail("CSR must be a JSON object")
    if csr.get("kind") != "CertificateSigningRequest":
        fail(f"kind != CertificateSigningRequest: {csr.get('kind')}")

    spec = csr.get("spec")
    if not isinstance(spec, dict):
        fail("spec missing/not object")

    if expected_node is not None and expected_node != live_node:
        fail(f"expected node {expected_node} != live node {live_node}")

    if spec.get("signerName") != SIGNER:
        fail(f"signerName != {SIGNER}: {spec.get('signerName')}")

    username = spec.get("username", "")
    if username != f"system:node:{live_node}":
        fail(f"username != system:node:{live_node}: {username}")

    groups = spec.get("groups", [])
    if not isinstance(groups, list) or "system:nodes" not in groups:
        fail(f"groups missing system:nodes: {groups}")

    usages = spec.get("usages", [])
    if not isinstance(usages, list):
        fail("usages missing/not list")
    usages_set = set(usages)
    if "server auth" not in usages_set:
        fail("usages missing 'server auth'")
    if usages_set - ALLOWED_USAGES:
        fail(f"unexpected usages: {usages_set - ALLOWED_USAGES}")

    request_b64 = spec.get("request", "")
    if not request_b64:
        fail("spec.request empty")

    cn, o, dns_sans, ip_sans, other_sans = parse_csr(request_b64)

    if cn != f"system:node:{live_node}":
        fail(f"CN != system:node:{live_node}: {cn}")
    if o != "system:nodes":
        fail(f"O != system:nodes: {o}")

    if other_sans:
        fail(f"unexpected SAN types: {other_sans}")

    if dns_sans != [live_node]:
        fail(f"DNS SAN != [{live_node}]: {dns_sans}")
    if ip_sans != [live_ip]:
        fail(f"IP SAN != [{live_ip}]: {ip_sans}")

    print("POLICY_RESULT=PASS")
    print(f"NODE={live_node}")
    print(f"INTERNAL_IP={live_ip}")
    print("NODE_READY=YES")
    print(f"SIGNER={spec.get('signerName')}")
    print(f"USERNAME={username}")
    print(f"CN={cn}")
    print(f"O={o}")
    print(f"DNS_SANS={dns_sans}")
    print(f"IP_SANS={ip_sans}")
    print(f"OTHER_SAN_TYPES={other_sans}")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("csr_json")
    ap.add_argument("--node-json", required=True)
    ap.add_argument("--expected-node", default=None)
    args = ap.parse_args()

    live_node, live_ip = load_node(args.node_json)
    with open(args.csr_json) as f:
        csr = json.load(f)

    try:
        validate(csr, live_node, live_ip, args.expected_node)
    except AssertionError as e:
        print("POLICY_RESULT=FAIL")
        print(f"REASON={e}")
        sys.exit(1)
    sys.exit(0)


if __name__ == "__main__":
    main()
