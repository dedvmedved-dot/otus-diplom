---
# TASK-P7B1-R1 — Hardened operator-invoked kubelet-serving CSR approval.
# Node JSON is AUTHORITATIVE for InternalIP. Operator supplies ONLY csr_name,
# node_name and approve flag (default false). No operator-supplied node_ip.
# No wildcard, no approve-all, no daemon/CronJob.

- name: "P7B1-R1 approve kubelet serving CSR (hardened, controlled)"
  hosts: node-01
  gather_facts: false
  vars:
    csr_name: "{{ csr_name | mandatory }}"
    node_name: "{{ node_name | mandatory }}"
    approve: "{{ approve | default(false) }}"

  tasks:
    - name: Fetch exact live Node JSON (authoritative)
      ansible.builtin.shell:
        cmd: |
          kubectl --kubeconfig /etc/kubernetes/admin.conf get node "{{ node_name }}" -o json > /tmp/p7b1-node-"{{ node_name }}".json
          kubectl --kubeconfig /etc/kubernetes/admin.conf get node "{{ node_name }}" -o jsonpath='{.status.addresses[?(@.type=="InternalIP")].address}'
        executable: /bin/bash
      register: live_node
      changed_when: false

    - name: Fetch exact CSR JSON
      ansible.builtin.shell:
        cmd: |
          kubectl --kubeconfig /etc/kubernetes/admin.conf get csr "{{ csr_name }}" -o json > /tmp/p7b1-csr-"{{ csr_name }}".json
          echo "fetched csr {{ csr_name }}"
        executable: /bin/bash
      changed_when: false

    - name: Copy policy checker to node
      ansible.builtin.copy:
        src: "../tools/p7b1_validate_kubelet_serving_csr.py"
        dest: /tmp/p7b1_validate_kubelet_serving_csr.py
        mode: "0755"

    - name: Validate CSR policy against live Node (FAIL-CLOSED)
      ansible.builtin.shell:
        cmd: |
          python3 /tmp/p7b1_validate_kubelet_serving_csr.py \
            /tmp/p7b1-csr-"{{ csr_name }}".json \
            --node-json /tmp/p7b1-node-"{{ node_name }}".json \
            --expected-node "{{ node_name }}"
        executable: /bin/bash
      register: policy
      changed_when: false

    - name: Approve EXACTLY this CSR (only policy PASS AND approve=true)
      ansible.builtin.shell:
        cmd: |
          kubectl --kubeconfig /etc/kubernetes/admin.conf certificate approve "{{ csr_name }}"
          kubectl --kubeconfig /etc/kubernetes/admin.conf get csr "{{ csr_name }}" -o jsonpath='{.status.conditions[?(@.type=="Approved")].status}'
        executable: /bin/bash
      register: approval
      when:
        - policy.rc == 0
        - approve | bool
      changed_when: true

    - name: Report dry-run (no approval) status
      ansible.builtin.debug:
        msg: "approve={{ approve }} -> approval NOT executed (dry-run / policy gate)"
      when: not (approve | bool) or policy.rc != 0
