# TASK-P7B1-R1 — Hardened fail-closed P7B1 final read-only verifier.
# P6D/P7A preservation markers are emitted ONLY after real canonical verifier runs.
# MODE: STRICTLY READ-ONLY.

- name: "P7B1 final read-only verifier (hardened)"
  hosts: all
  gather_facts: false

  tasks:
    - name: P7B1 cluster semantic verification (read-only)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import json, subprocess
          KC = ["kubectl", "--kubeconfig", "/etc/kubernetes/admin.conf"]
          def runj(c):
              r = subprocess.run(c, capture_output=True, text=True)
              if r.returncode != 0:
                  raise AssertionError(f"query failed: {' '.join(c)} rc={r.returncode}")
              return json.loads(r.stdout)

          nodes = runj(KC + ["get", "nodes", "-o", "json"])
          items = nodes.get("items")
          assert isinstance(items, list), "nodes invalid items"
          for it in items:
              assert it["status"]["nodeInfo"]["kubeletVersion"] == "v1.36.2", "k8s drift"
          print("P7B1_K8S_OK")
          assert len(items) == 3, f"node count {len(items)}"
          for it in items:
              ready = any(c["type"]=="Ready" and c["status"]=="True" for c in it["status"]["conditions"])
              assert ready, f"{it['metadata']['name']} not Ready"
          print("P7B1_NODES_READY")

          mc = runj(KC + ["-n","metallb-system","get","deploy","controller","-o","json"])
          assert mc["status"].get("availableReplicas",0)==1, "metallb"
          ms = runj(KC + ["-n","metallb-system","get","ds","speaker","-o","json"])
          assert ms["status"].get("numberReady",0)==3, "metallb speaker"
          print("P7B1_P6B_PRESERVED")

          ed = runj(KC + ["-n","envoy-gateway-system","get","deploy","envoy-gateway","-o","json"])
          assert ed["status"].get("availableReplicas",0)==1, "envoy"
          gc = runj(KC + ["get","gatewayclass","eg","-o","json"])
          acc=[c["status"] for c in gc["status"]["conditions"] if c["type"]=="Accepted"]
          assert acc and acc[0]=="True", "gatewayclass"
          print("P7B1_P6C_PRESERVED")

          csrs = runj(KC + ["get","csr","-o","json"])
          csritems = csrs.get("items")
          assert isinstance(csritems, list), "csr invalid items"
          pending = [c for c in csritems if c["spec"].get("signerName")=="kubernetes.io/kubelet-serving"
                     and not any(cond.get("type") in ("Approved","Denied") for cond in c["status"].get("conditions",[]))]
          assert len(pending)==0, f"unexpected pending serving CSR: {len(pending)}"
          print("P7B1_CSR_STATE_OK")
          PYEOF
      changed_when: false
      delegate_to: node-01
      run_once: true

    - name: P7B1 node kubelet TLS + config verification (read-only, all nodes)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import subprocess, socket, os, re
          host = socket.gethostname()
          cfg = open("/var/lib/kubelet/config.yaml").read()
          assert "serverTLSBootstrap: true" in cfg, "serverTLSBootstrap not true"
          print("P7B1_SERVER_TLS_BOOTSTRAP_OK")
          assert "rotateCertificates: true" in cfg, "rotateCertificates not true"
          print("P7B1_ROTATE_CERTIFICATES_OK")
          cert = None
          for c in ["/var/lib/kubelet/pki/kubelet-server-current.pem", "/var/lib/kubelet/pki/kubelet.crt"]:
              if os.path.exists(c): cert = c; break
          assert cert, "no serving cert"
          v = subprocess.run(["openssl","verify","-CAfile","/etc/kubernetes/pki/ca.crt",cert],capture_output=True,text=True)
          assert v.returncode==0, f"leaf not trusted: {v.stdout} {v.stderr}"
          print("P7B1_KUBELET_CLUSTER_CA_TRUST_OK")
          san = subprocess.run(["openssl","x509","-in",cert,"-noout","-ext","subjectAltName"],capture_output=True,text=True).stdout
          assert host in san, f"DNS SAN missing {host}"
          print("P7B1_KUBELET_DNS_SAN_OK")
          p = subprocess.run(["bash","-c","tr '\\0' '\\n' < /proc/$(pgrep -xo kubelet)/cmdline | grep -oE 'node-ip=[0-9.]+' | cut -d= -f2"],capture_output=True,text=True)
          ip = p.stdout.strip()
          assert ip and ip in san, f"IP SAN missing {ip}"
          print("P7B1_KUBELET_IP_SAN_OK")
          t = subprocess.run(["openssl","s_client","-connect",f"{ip}:10250","-verify_return_error","-CAfile","/etc/kubernetes/pki/ca.crt","-verify_ip",ip],input="",capture_output=True,text=True)
          assert t.returncode==0, f"InternalIP TLS failed"
          print("P7B1_KUBELET_TLS_INTERNALIP_OK")
          h = subprocess.run(["openssl","s_client","-connect",f"{ip}:10250","-servername",host,"-verify_hostname",host,"-verify_return_error","-CAfile","/etc/kubernetes/pki/ca.crt"],input="",capture_output=True,text=True)
          assert h.returncode==0, f"hostname TLS failed"
          print("P7B1_KUBELET_TLS_HOSTNAME_OK")
          PYEOF
      changed_when: false

    - name: P6D/P7A preservation via canonical verifiers (read-only, local)

          SSHPASS_VAL=$(cat /etc/hermes/pak_root_pass.txt)

          if ansible-playbook -i inventory/hosts.yaml -f 3 -e "ansible_ssh_pass=${SSHPASS_VAL}" -e "ansible_ssh_retries=10" ansible/playbooks/p6d-final-readonly-verify.yml -v > /tmp/p7b1-p6d-check.txt 2>&1; then

          fi
          if ansible-playbook -i inventory/hosts.yaml -f 3 -e "ansible_ssh_pass=${SSHPASS_VAL}" -e "ansible_ssh_retries=10" ansible/playbooks/p7a-final-readonly-verify.yml -v > /tmp/p7b1-p7a-check.txt 2>&1; then
            grep -q 'P7A_FINAL_READONLY_VERIFIER_PASS' /tmp/p7b1-p7a-check.txt || { echo "P7A marker missing" >&2; exit 1; }
            echo "P7B1_P7A_PRESERVED"
          else
            echo "P7A verifier rc != 0" >&2; tail -20 /tmp/p7b1-p7a-check.txt >&2; exit 1
          fi
          echo "P7B1_FINAL_READONLY_VERIFIER_PASS"
        executable: /bin/bash
      changed_when: false
      delegate_to: localhost
      connection: local
      run_once: true
