# TASK-P6C-R1 — Durable fail-closed P6C read-only verifier (Envoy Gateway 1.9.1).
# MODE: STRICTLY READ-ONLY. RUNTIME MUTATION = 0.
# Asserts exact P6C Envoy Gateway 1.9.1 contract (k8s v1.36.2 compatible) +
# P6B preservation. Fail-closed: any query failure / semantic violation -> FAIL.

- name: "P6C final read-only verifier (TASK-P6C-R1)"
  hosts: all
  gather_facts: false
  vars:
    envoy_ns: envoy-gateway-system
    controller_digest: "sha256:2999d87c3a2b3e890e0ee1a2b11a159b60963aadb2cbb9f2c0a49e0197df3c72"
    envoyproxy_digest: "sha256:9e24a3b459b463dd37e116a0f46d2abfcb6a1cb9897d87e85cf2ee25edbe373b"

  tasks:
    - name: P6C final semantic verification (read-only)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import json, subprocess, sys
          KC = ["kubectl", "--kubeconfig", "/etc/kubernetes/admin.conf"]
          def runj(c):
              r = subprocess.run(c, capture_output=True, text=True)
              if r.returncode != 0:
                  raise AssertionError(f"query failed: {' '.join(c)} rc={r.returncode} stderr={r.stderr}")
              return json.loads(r.stdout)

          # ── A. Project baseline: Kubernetes v1.36.2 (exact) ──
          nodes = runj(KC + ["get", "nodes", "-o", "json"])
          kubelet_versions = [it["status"]["nodeInfo"]["kubeletVersion"] for it in nodes["items"]]
          for kv in kubelet_versions:
              assert kv == "v1.36.2", f"project baseline k8s violation: {kv}"
          print("P6C_K8S_VERSION_OK")

          # ── B. Project baseline: Envoy Gateway v1.9.1 (exact) ──
          dep = runj(KC + ["-n", "envoy-gateway-system", "get", "deploy", "envoy-gateway", "-o", "json"])
          img = dep["spec"]["template"]["spec"]["containers"][0]["image"]
          assert "v1.9.1" in img or "2999d87c3a2b3e890e0ee1a2b11a159b60963aadb2cbb9f2c0a49e0197df3c72" in img, f"project baseline envoy violation (not 1.9.1): {img}"
          assert "latest" not in img and "master" not in img, f"floating image {img}"
          print("P6C_ENVOY_VERSION_OK")

          # ── C. Compatibility policy (Envoy minor -> supported k8s minors) ──
          SUPPORTED_K8S_BY_ENVOY_MINOR = {
              "1.9": {"1.33", "1.34", "1.35", "1.36"},
          }
          def _minor(v):
              parts = v.lstrip("v").split(".")
              return parts[0] + "." + parts[1]
          envoy_minor = _minor("v1.9.1")
          if envoy_minor not in SUPPORTED_K8S_BY_ENVOY_MINOR:
              raise AssertionError(f"no compatibility policy for Envoy minor {envoy_minor}")
          supported = SUPPORTED_K8S_BY_ENVOY_MINOR[envoy_minor]
          for kv in kubelet_versions:
              k8s_minor = _minor(kv)
              assert k8s_minor in supported, f"k8s {kv} not in Envoy {envoy_minor} supported set {sorted(supported)}"
          print("P6C_COMPATIBILITY_OK")

          # C. controller Ready + digest
          assert dep["status"].get("availableReplicas", 0) == dep["spec"]["replicas"] == 1, "controller not ready"
          ctrl_pods = runj(KC + ["-n", "envoy-gateway-system", "get", "pods", "-l", "app.kubernetes.io/name=gateway-helm", "-o", "json"])
          ctrl_imgid = ctrl_pods["items"][0]["status"]["containerStatuses"][0]["imageID"]
          assert "sha256:2999d87c3a2b3e890e0ee1a2b11a159b60963aadb2cbb9f2c0a49e0197df3c72" in ctrl_imgid, f"controller imageID {ctrl_imgid}"
          print("P6C_CONTROLLER_READY")
          print("P6C_CONTROLLER_IMAGE_OK")

          # D. required CRDs Established
          for crd in ["gatewayclasses.gateway.networking.k8s.io",
                      "gateways.gateway.networking.k8s.io",
                      "httproutes.gateway.networking.k8s.io",
                      "referencegrants.gateway.networking.k8s.io",
                      "envoyproxies.gateway.envoyproxy.io"]:
              obj = runj(KC + ["get", "crd", crd, "-o", "json"])
              est = [c["status"] for c in obj["status"]["conditions"] if c["type"] == "Established"]
              assert est and est[0] == "True", f"crd {crd} not Established"
          print("P6C_GATEWAY_API_CRDS_OK")

          # E. GatewayClass eg Accepted=True
          gc = runj(KC + ["get", "gatewayclass", "eg", "-o", "json"])
          acc = [c["status"] for c in gc["status"]["conditions"] if c["type"] == "Accepted"]
          assert acc and acc[0] == "True", "GatewayClass eg not Accepted"
          print("P6C_GATEWAYCLASS_ACCEPTED")

          # F/G/H. residue zero — fail-closed (query failure -> raise)
          for ns in ["p6c-validation", "p6c-r1-validation"]:
              r = subprocess.run(KC + ["get", "namespace", ns], capture_output=True, text=True)
              assert r.returncode != 0 and "NotFound" in r.stderr, f"validation ns {ns} still present"

          # Gateway residue (fail-closed)
          gw = runj(KC + ["get", "gateway", "-A", "-o", "json"])
          assert isinstance(gw.get("items"), list), "gateway query invalid JSON"
          assert len(gw["items"]) == 0, f"temporary Gateways remain: {len(gw['items'])}"
          print("P6C_GATEWAY_RESIDUE_ZERO")

          # HTTPRoute residue (fail-closed)
          hr = runj(KC + ["get", "httproute", "-A", "-o", "json"])
          assert isinstance(hr.get("items"), list), "httproute query invalid JSON"
          assert len(hr["items"]) == 0, f"temporary HTTPRoutes remain: {len(hr['items'])}"
          print("P6C_HTTPROUTE_RESIDUE_ZERO")

          # LB residue
          all_svc = runj(KC + ["get", "svc", "-A", "-o", "json"])
          lb = [s for s in all_svc["items"] if s["spec"].get("type") == "LoadBalancer"]
          assert len(lb) == 0, f"temporary LB services remain: {[s['metadata']['name'] for s in lb]}"
          print("P6C_LB_RESIDUE_ZERO")
          print("P6C_VALIDATION_RESIDUE_ZERO")

          # P6B preserved (BGP/BFD zero, label absent, kube-vip)
          for kind, marker in [
              ("bgppeer", "P6C_BGPPEER_ZERO"),
              ("bgpadvertisement", "P6C_BGPADVERTISEMENT_ZERO"),
              ("bfdprofile", "P6C_BFDPROFILE_ZERO"),
          ]:
              r = subprocess.run(KC + ["-n", "metallb-system", "get", kind, "-o", "json"], capture_output=True, text=True)
              assert r.returncode == 0, f"{kind} query failed rc={r.returncode}"
              obj = json.loads(r.stdout)
              assert isinstance(obj.get("items"), list), f"{kind} invalid JSON"
              assert len(obj["items"]) == 0, f"{kind} nonzero"
              print(marker)
          for n in ["node-01", "node-02", "node-03"]:
              node = runj(KC + ["get", "node", n, "-o", "json"])
              assert "node.kubernetes.io/exclude-from-external-load-balancers" not in node["metadata"]["labels"], f"{n} exclusion label present"
          kvp = runj(KC + ["-n", "kube-system", "get", "pod", "kube-vip-node-01", "-o", "json"])
          env = {e["name"]: e.get("value", "") for e in kvp["spec"]["containers"][0]["env"]}
          assert env.get("svc_enable") == "false" and env.get("address") == "172.30.140.100", "kube-vip regression"
          print("P6C_P6B_PRESERVED")

          # no floating images
          assert "@sha256:" in img, "controller image not digest-pinned"
          print("P6C_NO_FLOATING_IMAGES")

          print("P6C_FINAL_READONLY_VERIFIER_PASS")
          PYEOF
      changed_when: false
      delegate_to: node-01
      run_once: true

    # images.lock + versions.lock exactness (repository files, localhost)
    - name: P6C images.lock + versions.lock exactness (read-only)
      ansible.builtin.shell:
        cmd: |
          set -e
          python3 << 'PYEOF'
          import yaml, sys
          repo = "{{ playbook_dir }}/../.."
          doc = yaml.safe_load(open(repo + "/images.lock"))
          entries = doc.get("images", [])
          found = {e.get("component"): e for e in entries if e.get("component") in ("envoy-gateway", "envoy-proxy-dataplane")}
          assert "envoy-gateway" in found and "envoy-proxy-dataplane" in found, "envoy entries missing"
          assert found["envoy-gateway"]["tag"] == "v1.9.1", "envoy-gateway tag"
          assert found["envoy-gateway"]["digest"] == "sha256:2999d87c3a2b3e890e0ee1a2b11a159b60963aadb2cbb9f2c0a49e0197df3c72", "envoy-gateway digest"
          assert found["envoy-proxy-dataplane"]["digest"] == "sha256:9e24a3b459b463dd37e116a0f46d2abfcb6a1cb9897d87e85cf2ee25edbe373b", "envoy-proxy digest"
          assert found["envoy-gateway"].get("verified") is True, "envoy-gateway verified"
          assert found["envoy-proxy-dataplane"].get("verified") is True, "envoy-proxy verified"
          print("P6C_IMAGES_LOCK_OK")
          vl = yaml.safe_load(open(repo + "/versions.lock"))
          comps = vl.get("components", vl)
          assert comps.get("envoy_gateway") == "1.9.1", f"versions.lock envoy_gateway {comps.get('envoy_gateway')}"
          print("P6C_VERSIONS_LOCK_OK")
          PYEOF
      changed_when: false
      delegate_to: localhost
      run_once: true
